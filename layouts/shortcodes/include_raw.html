{{- $path := .Get "path" -}} {{- $raw := readFile $path -}} {{- /* Strip front matter if present */ -}} {{- $noFM := replaceRE `(?s)^---.*?---\s*` "" $raw -}} {{- /* Remove any stray or tags in content */ -}} {{- $noHead := replaceRE `(?is)]*>` "" $noFM -}} {{- /* Optionally strip legacy Next button hard-links */ -}} {{- $stripNext := .Get "strip_next" | default "true" -}} {{- $content := $noHead -}} {{- if eq $stripNext "true" -}} {{- $content = replaceRE `(?is)]*class=\"next-button\"[^>]*>.*?` "" $content -}} {{- end -}} {{- $content | safeHTML -}}
